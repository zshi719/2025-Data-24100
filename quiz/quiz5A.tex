conn = get_db_connection() # returns an SQL Connection

@app.route('/api/movies/sales/<movie_title>', methods=['GET'])
def movie_sales(movie_title):
    cursor = conn.cursor()
    cursor.execute(
        f"SELECT date, tickets_sold FROM sales WHERE movie = {movie_title}")
    results = cursor.fetchall()
    # Convert tickets to thousands
    formatted_results = [(date, tickets / 1000) for date, tickets in results]
    return jsonify({'movie': movie_title, 'sales': formatted_results}), 200

@app.route('/api/movies/revenue/<movie_title>', methods=['GET'])
